package com.example.shop;

import com.example.shop.api.*;
import com.example.shop.model.*;
import dev.openapifn.spring.FilePart;
import dev.openapifn.spring.client.OpenApiHttpClient;
import java.net.URI;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;
import org.springframework.web.client.RestClient;
import tools.jackson.databind.json.JsonMapper;

/** Calls every operation through the generated clients and describes the results. */
public final class EndToEnd {

    public static List<String> run(String baseUrl) {
        var http = new OpenApiHttpClient(RestClient.builder().defaultHeader("X-Api-Key", "secret").build(), baseUrl, JsonMapper.builder().build());
        var pets = new PetsClient(http);
        var orders = new OrdersClient(http);
        var defaults = new DefaultClient(http);
        var results = new ArrayList<String>();

        var listRequest = ListPetsRequest.builder().limit(2).tags(List.of("a", "b")).xRequestId(UUID.randomUUID()).session("s").build();
        var listed = pets.listPets(listRequest);
        if (listed instanceof ListPetsResponse.Ok ok) {
            results.add("list:" + ok.xTotal() + ":" + ok.body().stream().map(p -> p.getClass().getSimpleName() + "/" + p.name()).toList());
            var next = pets.next(listRequest, ok);
            results.add("next:" + next.limit() + ":" + next.tags() + ":" + next.xRequestId().equals(listRequest.xRequestId()) + ":" + next.session());
        }
        var root = baseUrl.substring(0, baseUrl.indexOf("/api/v1"));
        var createRequest = new CreatePetRequest(URI.create(root + "/callbacks/petCreated"), Dog.builder().name("Bello").kind(Kind.DOG).barks(false).build());
        var created = pets.createPet(createRequest);
        results.add(switch (created) {
            case CreatePetResponse.Created c -> "created:" + c.location() + ":" + c.body().getClass().getSimpleName() + "/" + c.body().id();
            case CreatePetResponse.Status4XX problem -> "problem:" + problem.status();
        });
        var createdPet = (CreatePetResponse.Created) created;
        @SuppressWarnings("deprecation")
        var deleteLinked = pets.deleteCreatedPet(createRequest, createdPet);
        results.add("linked:" + pets.getCreatedPet(createRequest, createdPet).petId() + ":" + deleteLinked.petId());
        var rejected = pets.createPet(new CreatePetRequest(null, new Pet.Default(null, "error", Kind.BIRD, null, null)));
        results.add(switch (rejected) {
            case CreatePetResponse.Created c -> "unexpected";
            case CreatePetResponse.Status4XX problem -> "problem:" + problem.status() + ":" + problem.body().detail();
        });
        var searched = pets.searchPets(new SearchPetsRequest(new SearchPetsRequest.Body("e")));
        results.add("search:" + ((SearchPetsResponse.Ok) searched).body().size());
        results.add("get:" + switch (pets.getPet(new GetPetRequest(2L, null))) {
            case GetPetResponse.Ok ok -> ok.body().name();
            case GetPetResponse.OkPlain plain -> plain.body();
            case GetPetResponse.NotFound ignored -> "missing";
        });
        results.add("get404:" + (pets.getPet(new GetPetRequest(99L, null)) instanceof GetPetResponse.NotFound));
        @SuppressWarnings("deprecation")
        var deleted = pets.deletePet(new DeletePetRequest(1L));
        results.add("delete:" + deleted.status());
        var photo = pets.uploadPhoto(new UploadPhotoRequest(1L, new UploadPhotoRequest.Body(FilePart.of("p.png", "image/png",
            "PNG".getBytes(StandardCharsets.UTF_8)), "caption", new UploadPhotoRequest.Body.Meta(10))));
        var file = ((UploadPhotoResponse.Ok) photo).body();
        results.add("photo:" + file.contentType() + ":" + file.text());
        var orderRequest = new GetOrderRequest(EndToEndIds.ORDER, List.of(GetOrderRequest.ExpandItem.ITEMS));
        var orderResponse = (GetOrderResponse.Status2XX) orders.getOrder(orderRequest);
        var note = orders.addOrderNote(orders.note(orderRequest, orderResponse));
        results.add("links:" + orders.receipt(orderRequest, orderResponse).orderId().equals(EndToEndIds.ORDER) + ":" + ((AddOrderNoteResponse.Ok) note).body());
        var order = ((GetOrderResponse.Status2XX) orders.getOrder(new GetOrderRequest(EndToEndIds.ORDER, List.of(GetOrderRequest.ExpandItem.ITEMS)))).body();
        results.add("order:" + order.status() + ":" + order.priority() + ":" + order.payment() + ":" + order.shipping().getClass().getSimpleName()
            + ":" + order.lines());
        var pickup = ((GetOrderResponse.Status2XX) orders.getOrder(new GetOrderRequest(EndToEndIds.ORDER, List.of(GetOrderRequest.ExpandItem.CUSTOMER)))).body();
        results.add("pickup:" + pickup.shipping());
        try (var events = ((OrderEventsResponse.Ok) orders.orderEvents()).body()) {
            results.add("events:" + events.map(OrderEvent::type).toList());
        }
        var imported = orders.importEvents(new ImportEventsRequest(java.util.stream.Stream.of(new OrderEvent("placed", EndToEndIds.ORDER),
            new OrderEvent("shipped", EndToEndIds.ORDER))));
        results.add("imported:" + ((ImportEventsResponse.Ok) imported).body());
        var receipt = orders.getReceipt(new GetReceiptRequest(EndToEndIds.ORDER, null));
        var receiptText = orders.getReceipt(new GetReceiptRequest(EndToEndIds.ORDER, dev.openapifn.spring.Accept.of("text/plain")));
        results.add("receipt:" + ((GetReceiptResponse.Ok) receipt).body().total() + ":" + ((GetReceiptResponse.OkPlain) receiptText).body());
        var jsonNote = orders.addOrderNote(new AddOrderNoteRequest(EndToEndIds.ORDER, new AddOrderNoteRequest.Body("hi"), null));
        var textNote = orders.addOrderNote(new AddOrderNoteRequest(EndToEndIds.ORDER, null, "ho"));
        results.add("notes:" + ((AddOrderNoteResponse.Ok) jsonNote).body() + ":" + ((AddOrderNoteResponse.Ok) textNote).body());
        results.add("health:" + ((HealthResponse.Ok) defaults.health()).body());

        // Webhooks and callbacks are sent to receivers; here the same application receives them.
        var webhooks = new WebhooksClient(http);
        var callbacks = new CallbacksClient(http);
        var tom = Cat.builder().name("Tom").kind(Kind.CAT).build();
        results.add("adopted:" + webhooks.petAdopted(URI.create(root + "/webhooks/petAdopted"), new PetAdoptedRequest("signed", tom)).status());
        results.add("stock:" + webhooks.postStockLow(URI.create(root + "/webhooks/postStockLow"),
            new PostStockLowRequest(new PostStockLowRequest.Body("sku-1"))).status());
        // Sent to the same URL with the same method as petCreatedCallback, and told apart by its body.
        results.add("rejected:" + callbacks.petRejected(callbacks.petRejectedUrl(createRequest), new PetRejectedRequest(new Problem("Rejected", 422,
            "no more cats"))).status());
        results.add("callback:" + callbacks.petCreatedCallback(callbacks.petCreatedCallbackUrl(createRequest), new PetCreatedCallbackRequest(tom))
            .status());
        results.add("put:" + ((PetCreatedPutResponse.Ok) callbacks.petCreatedPut(URI.create(root + "/callbacks/petCreated"),
            new PetCreatedPutRequest("hi"))).body());
        try {
            callbacks.petCreatedCallback(URI.create(root + "/callbacks/petCreated"), new PetCreatedCallbackRequest(new Pet.Default(null, "", Kind.BIRD,
                null, null)));
            results.add("invalid:accepted");
        } catch (dev.openapifn.spring.client.UnexpectedResponseException e) {
            results.add("invalid:" + e.response().status());
        }
        return results;
    }

    private EndToEnd() {
    }
}
